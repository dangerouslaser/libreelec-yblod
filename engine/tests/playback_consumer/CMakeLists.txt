cmake_minimum_required(VERSION 3.20)
project(YblodInstalledPlaybackConsumer LANGUAGES C)
find_package(YblodNative 0.1 CONFIG REQUIRED)
if(NOT TARGET Yblod::yblod_playback_native)
  message(FATAL_ERROR "Installed YblodNative lacks the opt-in experimental playback profile")
endif()
add_executable(native_playback_consumer consumer.c)
target_compile_features(native_playback_consumer PRIVATE c_std_11)
target_compile_options(native_playback_consumer PRIVATE -Wall -Wextra -Werror -fno-lto)
target_link_libraries(native_playback_consumer PRIVATE Yblod::yblod_playback_native)
